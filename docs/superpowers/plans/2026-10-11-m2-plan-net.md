### 0.2 NGO facts checked in package source (`Library/PackageCache/com.unity.netcode.gameobjects@277a288a1cea/Runtime`)
These close or narrow skeleton risks 2 and 5. The T19 PlayMode tests stay the runtime check.

| Id | Fact | Source |

| N1 | `SendTo.Owner` on an object the local peer owns (solo, or a host-owned rod) goes through `LocalSendRpcTarget`. With `DeferLocal = true`, the call is queued as `OnNextFrame` and dispatched at the **next frame's `EarlyUpdate`**, before any script `Update` in that frame. **`CastProbe`:** C 7.17. Uses a static `RaycastHit[16]` buffer.
1. Ray 1 starts at `(x, WaterSurface.FlatHeight + FishingConstants.WaterProbeHeight, z)`, goes down, with length `WaterProbeHeight + BedProbeDistance`, `CastMask`, `QueryTriggerInteraction.Collide`.
2. Pick the smallest `distance` (`RaycastNonAlloc` is unsorted).
3. If its collider's `gameObject.layer != Layers.Water`, the result has `FirstHitIsWater = false`.
4. Otherwise set `WaterY = hit.point.y` and cast ray 2 from `hit.point` down `BedProbeDistance` against `BedMask` with `QueryTriggerInteraction.Ignore`. On a hit, set `HasBed = true` and `Depth = WaterY - bed.point.y`.
- `CastMask` = Water | Ground | Shore | Structure. `BedMask` = Ground.
- V12 is checked by T19's PlayMode probe asserts.

### RPC sequences (C 8.2, C 9) - code
Every attribute sets `InvokePermission`. Server handlers forward to `FishingHost`. With no host, the rod replies so the owner never stalls.
```csharp
// ---------------- FishingRodNetwork: owner -> host intents ----------------
// On a host-owned rod these run synchronously inside the caller (N2).
[Rpc(SendTo.Server, InvokePermission = RpcInvokePermission.Owner)]
public void CastRequestRpc(uint castId, Vector3 origin, float yawDeg, float chargeNorm, byte baitIndex, bool assist, RpcParams p = default)
{
    var req = new CastRequestData { CastId = castId, OriginX = origin.x, OriginY = origin.y, OriginZ = origin.z,
                                    YawDeg = yawDeg, ChargeNorm = chargeNorm, BaitIndex = baitIndex, Assist = assist };
    FishingHost host = ResolveHost();
    if (host == null) { CastResultRpc(castId, false, (byte)CastRejectReason.NotAllowedNow, origin, default); return; }
    host.HandleCastRequest(this, in req);
}

[Rpc(SendTo.Server, InvokePermission = RpcInvokePermission.Owner)]
public void CancelRpc(uint castId, RpcParams p = default)
{
    FishingHost host = ResolveHost();
    if (host == null) { CommitRpc(castId, false, (byte)FightOutcome.Cancelled, (byte)EscapeReason.None); return; }
    host.HandleCancel(this, castId);
}

[Rpc(SendTo.Server, InvokePermission = RpcInvokePermission.Owner)]
public void HookedRpc(uint castId, RpcParams p = default)
{
    FishingHost host = ResolveHost();
    if (host == null) { CommitRpc(castId, false, (byte)FightOutcome.Escaped, (byte)EscapeReason.Invalid); return; }
    host.HandleHooked(this, castId);
}

[Rpc(SendTo.Server, InvokePermission = RpcInvokePermission.Owner)]
public void ResultRpc(uint castId, byte outcome, byte reason, float maxTension, float fightSeconds, ushort nearSnapRecoveries, RpcParams p = default)
{
    FishingHost host = ResolveHost();
    if (host == null) { CommitRpc(castId, false, outcome, reason); return; }
    var report = new FightReport { MaxTension = maxTension, FightSeconds = fightSeconds, NearSnapRecoveries = nearSnapRecoveries };
    host.HandleResult(this, castId, (FightOutcome)outcome, (EscapeReason)reason, in report);
}

[Rpc(SendTo.Server, InvokePermission = RpcInvokePermission.Owner)]

internal void HandleCastRequest(FishingRodNetwork rod, in CastRequestData req)
{

    if (!EnsureLedger() || rod.Player == null || !rod.Player.IsServerInitialized)
    {
        rod.CastResultRpc(req.CastId, false, (byte)CastRejectReason.UnknownPlayer, origin, default);

    Vector3 seen = rod.transform.position;                     // host's interpolated view of the sender's root
    var view = new HostCastView { PlayerX = seen.x, PlayerY = seen.y, PlayerZ = seen.z, CastingAllowed = true };
    if (Ledger.TryComputeLanding(slot, in req, out float lx, out float lz)) view.Probe = CastProbe.Probe(lx, lz);
    CastVerdict v = Ledger.RequestCast(rod.OwnerClientId, slot, in req, in view, FishingDebugSettings.ToOverrides(_registry));
    MirrorLoadout(rod, slot);                                  // bait consumed on accept -> NetSelectedBaitCount
    float y = v.Valid ? view.Probe.WaterY : WaterSurface.FlatHeight;

    if (v.Valid && v.Roll.Pity)                                // R65: the crew sees the lucky line (toast.pity, T25)
        rod.CatchAnnouncedRpc(slot, default, (byte)AnnounceKind.PityCast, 0);
}

internal void HandleCancel(FishingRodNetwork rod, uint castId)
{

    Commit(rod, Ledger.Cancel(rod.OwnerClientId, castId));     // Waiting: refund + clear; orphan: stored resolution
}

internal void HandleHooked(FishingRodNetwork rod, uint castId)
{
    if (!EnsureLedger()) { rod.CommitRpc(castId, false, (byte)FightOutcome.Escaped, (byte)EscapeReason.Invalid); return; }
    ulong client = rod.OwnerClientId;
    float hostDistance = 0f;
    if (Ledger.TryGetActive(client, out ActiveCast cast))

        float dx = p.x - cast.LandingX, dz = p.z - cast.LandingZ;

}

internal void HandleResult(FishingRodNetwork rod, uint castId, FightOutcome outcome, EscapeReason reason, in FightReport report)

// C 7.17 commit path: CommitRpc -> creel add -> CatchAnnouncedRpc -> FishCaught. CatchLog is kept by the ledger itself.
// Ruling E-Q6: Accepted is false ONLY for orphan repeats, so effects are driven by the decision fields, not by Accepted:
// creel add iff AddToCreel; announce iff Announce != None; FishCaught iff Outcome == Landed && AddToCreel. An orphan repeat
// carries AddToCreel false and Announce None, so it only unlocks the owner. Invalid and cancel resolutions announce nothing by design.
private void Commit(FishingRodNetwork rod, in CommitDecision d)
{
    rod.CommitRpc(d.CastId, d.Accepted, (byte)d.Outcome, (byte)d.Reason);       // 1. owner unlock (deferred on a host-owned rod)
    byte slot = rod.Player.Slot;
    {
        ref readonly FishingTuning t = ref _db.Tuning;
        if (d.AddToCreel)                                                        // 2. host creel (slot ledger + NetCreel mirror)
        {
            var item = new CreelItem { ItemIndex = d.Roll.Catch.ItemIndex, IsJunk = d.Roll.Catch.IsJunk, WeightG = d.Roll.Catch.WeightG,
                                       Quality = d.Roll.Catch.Quality, ValueCoins = d.Roll.Catch.ValueCoins, CaughtAtRunMinute = 0 };
            if (rod.Creel == null || !rod.Creel.ServerTryAdd(in item, t.CreelMaxEntries, t.CreelMaxWeightG, out _))
                Debug.LogError($"FishingHost: creel refused a catch the ledger accepted (slot {slot}).", this);   // same-frame snapshot: a bug
        }
        if (d.Announce != AnnounceKind.None)                                     // 3. everyone's reveal / toast
            rod.CatchAnnouncedRpc(slot, CatchResultNet.From(in d.Roll.Catch), (byte)d.Announce,
                                  d.CleanBonus ? RodWireQuant.AnnounceFlagClean : (byte)0);
        if (d.Outcome == FightOutcome.Landed && d.AddToCreel)                    // 4. host bus, published only on the host (not for ReleasedCreelFull)
        {
            EventBus bus = ResolveBus();
            if (bus != null) bus.Publish(new FishCaught(slot, in d.Roll.Catch));
        }
    }
    MirrorLoadout(rod, slot);                                                    // 5. refunds show at once
}

private void Update()
{
    if (Ledger == null || !IsServerActive) return;
    _unsolicited.Clear();
    Ledger.Tick(Time.deltaTime, IsPaused(), _unsolicited);   // pause-aware clock; stuck-Waiting refund; MaxFightTime -> Escaped(Timeout)
    for (int i = 0; i < _unsolicited.Count; i++)
    {
        CommitDecision d = _unsolicited[i];
        FishingRodNetwork rod = FindRod(d.ClientId);         // FishingRodNetwork.Active, OwnerClientId match
        if (rod != null) Commit(rod, in d);                  // null: the client left; Disconnect already cleared it
    }
}

After each `Tick` / `TickFight` / `OnCastResult` / `OnCommit` the controller compares it with the last value; when it grew and `Debug.isDebugBuild`, it calls `Debug.Log` (not LogWarning) once per state through a `LogOnce` keyed by the state name. It never warns on the reply to its own cancel (the machine consumes that silently), so ordinary cancels stay warning-free in PlayMode.
- **Host replies:** `OnCastResult(castId, valid, reason, landing, roll)` does the following:
  1. `profile` = `_db.Profiles[_db.Species[roll.Catch.ItemIndex].FightProfileIndex]` when `valid && !IsJunk`, else `default`.
  2. On an invalid result, `Debug.Log` the reason in debug builds. It never sends `CancelRpc` (BF-5): for a host-rejected result there is nothing to refund, and for an insane roll the `CastResultRpc` handler (above) has already sent the single `CancelRpc` before raising `CastResultReceived(valid:false)`. This method only reacts by calling `Machine.OnCastResult(castId, false, ...)`.
  3. `Machine.OnCastResult(castId, valid, in roll, in profile)`, then `ProcessEffects()`.
  - `OnCommit` calls `Machine.OnCommit(...)` and then `ProcessEffects()`.
  - Both arrive from deferred RPCs in `EarlyUpdate` (N1), never inside `Update`.

```csharp
private void Update()
{
    if (Machine == null) return;
    float dt = Time.deltaTime;
    bool paused = IsPaused();                                   // ServiceRoots.Game IPauseState (solo only)
    if (!paused) _clock += dt;                                  // unpaused owner clock (telemetry)
    PlayerIntent intent = _input.Intent;                        // routed with the context locomotion wrote last frame

    if (!paused)
    {
        HandleEquip(in intent, _input.GameplayContext);
        HandleSwim();
        ApplyRod(false);
        if (Machine.State == RodState.Idle) Machine.Assist = _settings.Current != null && _settings.Current.fishingAssist;
    }

    RodFrame frame = BuildFrame(in intent, paused);
    Machine.Tick(dt, in frame);                                 // all non-fight rows of spec 9.2; Paused freezes timers
    ProcessEffects();

    if (Machine.State == RodState.Hooked && !paused)
    {
        _fightAccumulator += dt;
        int steps = 0;
        while (_fightAccumulator >= FishingConstants.FightStep && steps < FishingConstants.MaxFightStepsPerFrame
               && Machine.State == RodState.Hooked)
        {
            _lastFightInput = _input.Fight;                     // RodFighting: (Steer, Reel || PrimaryHeld)
            frame.HookDisplacement = Flat(transform.position, _hookRoot);
            Machine.TickFight(in _lastFightInput, in frame);
            _fightAccumulator -= FishingConstants.FightStep;
            steps++;
            ProcessEffects();
        }
        if (steps == FishingConstants.MaxFightStepsPerFrame) _fightAccumulator = 0f;   // spec 9.1: extra time is dropped
    }
    else if (Machine.State != RodState.Hooked)
    {
        _fightAccumulator = 0f;
    }

    PublishView();
}

private RodFrame BuildFrame(in PlayerIntent intent, bool paused)
{
    var f = new RodFrame { Paused = paused, InWater = _locomotion.IsSwimming };
    if (RodOut && !paused)
    {
        f.PrimaryPressed = intent.PrimaryPressed;     f.PrimaryReleased = intent.PrimaryReleased;     f.PrimaryHeld = intent.PrimaryHeld;
        f.SecondaryPressed = intent.SecondaryPressed; f.SecondaryReleased = intent.SecondaryReleased; f.SecondaryHeld = intent.SecondaryHeld;
    }
    f.MarkerValid = UpdateMarker();                             // Ready/Charging only; false otherwise
    Vector3 root = transform.position;
    f.PlayerToBobber = _hasLanding ? Flat(root, _landing) : 0f;   // root -> bobber; R67: the machine subtracts CastTipReach for the maxLineOut check (limit measured from the rod tip)
    f.TipToBobber = _hasLanding ? Flat(RodTipPosition, _landing) : 0f;
    f.HookDisplacement = Machine.State == RodState.Hooked ? Flat(root, _hookRoot) : 0f;
    return f;
}

// Drains into a scratch list first: an effect may run a host handler synchronously on a host-owned rod (N2).
// Replies to the owner are deferred (N1), so no machine call re-enters from inside this loop; the guard is insurance.
private void ProcessEffects()
{
    if (_draining) { _redrain = true; return; }
    _draining = true;
    try
    {
        do
        {
            _redrain = false;
            _effects.Clear();
            Machine.DrainEffects(_effects);
            for (int i = 0; i < _effects.Count; i++)
            {
                RodEffect e = _effects[i];
                switch (e.Kind)
                {
                    case RodEffectKind.Cue:
                        _net.RaiseLocalCue(e.Cue, e.Cue == RodCue.Snapped ? e.SizeClass : (byte)0);   // BF-1: machine's Cue(Snapped) effect carries Roll.SizeClass
                        break;
                    case RodEffectKind.SendCastRequest:
                    {
                        Vector3 root = transform.position;
                        float yaw = transform.eulerAngles.y;
                        CastMath.Landing(root.x, root.z, yaw, Machine.CastDistance, out float lx, out float lz);   // R67: CastDistance is from the tip; Landing adds CastTipReach
                        _landing = new Vector3(lx, WaterSurface.FlatHeight, lz);   // set before the RPC: solo handles it inline
                        _hasLanding = true; _castYaw = yaw; _releaseAt = _clock; _reachedWaiting = false;
                        if (CanSend) _net.CastRequestRpc(e.CastId, root, yaw, e.ChargeNorm, _net.NetSelectedBait.Value, Machine.Assist);
                        break;
                    }
                    case RodEffectKind.SendCancel:
                        if (CanSend) _net.CancelRpc(e.CastId);
                        break;
                    case RodEffectKind.SendHooked:
                        if (CanSend) _net.HookedRpc(e.CastId);
                        break;
                    case RodEffectKind.SendResult:
                        if (CanSend) _net.ResultRpc(e.CastId, (byte)e.Outcome, (byte)e.Reason, e.Report.MaxTension, e.Report.FightSeconds,
                                                    (ushort)Mathf.Clamp(e.Report.NearSnapRecoveries, 0, ushort.MaxValue));
                        break;
                    case RodEffectKind.SendRodEvent:
                        if (CanSend) _net.RodEventRpc((byte)e.EventType, e.SizeClass);
                        break;
                    case RodEffectKind.StartFight:
                        _fightAccumulator = 0f; _hookRoot = transform.position; _sampler.Begin(_landing);
                        break;
                    case RodEffectKind.CommitTimedOut:
                        Debug.LogWarning($"FishingRodController: no CommitRpc for cast {e.CastId} within commitTimeout; returning to Idle.", this);
                        break;
                }
            }
            ObserveState();                                     // logs, rumble off at Idle only (BF-2: terminal pulses belong to FishingFeedback), telemetry capture, _hasLanding reset at Idle
        } while (_redrain);
    }
    finally { _draining = false; }
}

private void PublishView()
{
    RodState s = Machine.State;
    FightSim fight = Machine.Fight;
    byte flags = 0; float lateral = 0f, tension = 0f;
    if (fight != null)
    {
        ref readonly FightState fs = ref fight.State;
        lateral = fs.Lateral; tension = fs.Tension;
        if (s == RodState.Hooked)
        {
            if (fs.Phase == FightPhase.Burst) flags |= fs.BurstDir == 0 ? (byte)(RodWireQuant.FlagBurst | RodWireQuant.FlagDive) : RodWireQuant.FlagBurst;
            if (_lastFightInput.Reel) flags |= RodWireQuant.FlagReel;
            if (fight.Exhausted) flags |= RodWireQuant.FlagExhausted;
        }
    }
    if (Machine.Recalling) flags |= RodWireQuant.FlagRecall;
    Vector3 bobber = BobberPoint(s, fight, lateral);           // see "Bobber point"; sampler.Tick runs inside while Hooked
    _net.OwnerWriteState(new RodStateData
    {
        State = (byte)s, Flags = flags, LateralQ = RodWireQuant.QuantizeLateral(lateral),
        TensionQ = RodWireQuant.QuantizeTension(tension), BobberPos = bobber, CastId = Machine.CastId,
    });
    _net.OwnerSetView(new RodViewState { State = s, Flags = flags, Lateral = lateral, Tension = tension, BobberPos = bobber,
                                         CastId = Machine.CastId, IsOwner = true, RodOut = RodOut });
}
// CanSend => !_suppressSends && _net != null && _net.IsSpawned. Flat(a, b) = horizontal distance.
